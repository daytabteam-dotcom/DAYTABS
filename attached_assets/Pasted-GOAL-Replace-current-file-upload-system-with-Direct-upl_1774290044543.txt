2️⃣ CREATE R2 CLIENT
import { S3Client } from "@aws-sdk/client-s3";

const r2 = new S3Client({
  region: "auto",
  endpoint: `https://${process.env.R2_ACCOUNT_ID}.r2.cloudflarestorage.com`,
  credentials: {
    accessKeyId: process.env.R2_ACCESS_KEY_ID,
    secretAccessKey: process.env.R2_SECRET_ACCESS_KEY,
  },
});
3️⃣ GENERATE PRESIGNED UPLOAD URL
Endpoint: /generate-upload-url
import { PutObjectCommand } from "@aws-sdk/client-s3";
import { getSignedUrl } from "@aws-sdk/s3-request-presigner";

app.post("/generate-upload-url", async (req, res) => {
  const fileName = `videos/${Date.now()}.mp4`;

  const command = new PutObjectCommand({
    Bucket: process.env.R2_BUCKET_NAME,
    Key: fileName,
    ContentType: "video/mp4",
  });

  const uploadUrl = await getSignedUrl(r2, command, { expiresIn: 60 });

  res.json({
    uploadUrl,
    fileKey: fileName,
    fileUrl: `${process.env.R2_PUBLIC_URL}/${fileName}`,
  });
});

5️⃣ BACKEND: DOWNLOAD FROM R2 FOR PROCESSING

Inside your endpoints (/editing-assistant, etc):

import fs from "fs";
import fetch from "node-fetch";

async function downloadVideo(videoUrl) {
  const res = await fetch(videoUrl);
  const buffer = await res.arrayBuffer();
  fs.writeFileSync("input.mp4", Buffer.from(buffer));
}
Use it:
await downloadVideo(videoUrl);

// THEN run your existing ffmpeg pipeline
6️⃣ DELETE VIDEO AFTER PROCESSING
IMPORTANT: Always delete after report generation
import { DeleteObjectCommand } from "@aws-sdk/client-s3";

async function deleteFromR2(fileKey) {
  await r2.send(
    new DeleteObjectCommand({
      Bucket: process.env.R2_BUCKET_NAME,
      Key: fileKey,
    })
  );
}
Call it at END of each endpoint:
try {
  // process video
  // generate report

  res.json(result);

} finally {
  await deleteFromR2(fileKey);
}
